AI: Try this:
def summarise(data):
    cur = data.get("current_condition", [{}])[0]
    desc = cur.get("weatherDesc", [{}])[0].get("value", "N/A")
    temp = cur.get("temp_C", "N/A")
    wind = cur.get("windspeedKmph", "N/A")
    hum  = cur.get("humidity", "N/A")
    return f"Today: {desc}, {temp}°C, wind {wind} km/h, humidity {hum}%"
